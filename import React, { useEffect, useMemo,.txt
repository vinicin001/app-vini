        {section==="orcamentos"&&<div className="grid gap-4 sm:grid-cols-2">{Object.entries(data.budgets).map(([c,l])=>{const spent=byCategory[c]||0;return <Card key={c} className="p-5"><div className="flex justify-between"><div><b>{c}</b><p className="text-sm">Gasto: {money(spent)}</p></div><span>{l?Math.round(spent/l*100):0}%</span></div><input type="number" value={l} onChange={e=>setData(d=>({...d,budgets:{...d.budgets,[c]:Number(e.target.value)}}))} className="mt-4 w-full rounded-xl border p-3"/></Card>})}</div>}

        {section==="metas"&&<div className="space-y-4"><Card className="p-5"><p className="text-xs font-black text-[#367c2b]">META PRINCIPAL</p><select value={selectedGoalId} onChange={e=>setSelectedGoalId(Number(e.target.value))} className="mt-2 w-full rounded-2xl border p-3 font-bold">{data.goals.map(g=><option key={g.id} value={g.id}>{g.name}</option>)}</select></Card><div className="grid gap-4 sm:grid-cols-2">{data.goals.map(g=>{const p=Math.min(100,g.saved/g.target*100||0);return <Card key={g.id} className="p-5"><div className="flex justify-between"><div><p className="text-xs font-black text-[#367c2b]">META FINANCEIRA</p><h3 className="text-xl font-black">{g.name}</h3></div><span>{p.toFixed(0)}%</span></div><p className="mt-4 text-3xl font-black">{money(g.saved)}</p><div className="mt-3 h-3 overflow-hidden rounded-full bg-slate-100"><div className="h-full bg-[#367c2b]" style={{width:`${p}%`}}/></div><label className="mt-4 block text-xs font-bold text-slate-500">VALOR GUARDADO<input type="number" value={g.saved} onChange={e=>setData(d=>({...d,goals:d.goals.map(x=>x.id===g.id?{...x,saved:Number(e.target.value)}:x)}))} className="mt-1 w-full rounded-xl border p-3"/></label><label className="mt-3 block text-xs font-bold text-slate-500">PRAZO<input type="date" value={g.deadline||""} onChange={e=>setData(d=>({...d,goals:d.goals.map(x=>x.id===g.id?{...x,deadline:e.target.value}:x)}))} className="mt-1 w-full rounded-xl border p-3"/></label></Card>})}</div></div>}

        {section==="simulador"&&<div className="grid gap-5 lg:grid-cols-2"><Card className="p-6"><h2 className="text-2xl font-black">Simulador do carro</h2>{Object.entries(car).map(([k,v])=><label key={k} className="mt-3 block text-xs font-bold">{{price:"PREÇO",down:"ENTRADA",rate:"JUROS (%)",months:"MESES"}[k]}<input type="number" value={v} onChange={e=>setCar({...car,[k]:Number(e.target.value)})} className="mt-1 w-full rounded-xl border p-3"/></label>)}</Card><Card className="bg-gradient-to-br from-[#173d27] to-[#367c2b] p-6 text-white"><p>Parcela estimada</p><p className="mt-2 text-5xl font-black text-[#ffde00]">{money(installment)}</p><p className="mt-5">Custo mensal com uso: <b>{money(installment+500)}</b></p></Card></div>}
      </main>
    </div>

    <div className="fixed left-3 top-1/2 z-40 -translate-y-1/2"><div className="flex items-center gap-2"><button onClick={()=>setFloating(!floating)} className="grid h-14 w-14 place-items-center rounded-full bg-[#367c2b] text-white shadow-2xl ring-4 ring-white/80">{floating?<X/>:<Plus/>}</button>{floating&&<motion.div initial={{opacity:0,x:-10}} animate={{opacity:1,x:0}} className="w-52 rounded-2xl bg-white p-2 shadow-2xl"><button onClick={()=>{setSection("lancamentos");setFloating(false)}} className="w-full rounded-xl p-3 text-left font-bold hover:bg-slate-100">Lançar gasto</button><button onClick={()=>{setSection("calendario");setFloating(false)}} className="w-full rounded-xl p-3 text-left font-bold hover:bg-slate-100">Abrir calendário</button><button onClick={()=>{setSection("metas");setFloating(false)}} className="w-full rounded-xl p-3 text-left font-bold hover:bg-slate-100">Atualizar meta</button></motion.div>}</div></div>
  </div>;
}
